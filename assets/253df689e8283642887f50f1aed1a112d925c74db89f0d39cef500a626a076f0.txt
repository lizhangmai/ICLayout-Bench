.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp {temp_c}
.option klu gmin=1e-15 rshunt=1e15 reltol=1e-6 abstol=1e-15 vntol=1e-9 method=gear maxord=2
VIN src 0 pwl(0 0 20n 0 22n {sample_v} 400n {sample_v} 402n {hold_v} 11.2u {hold_v} 11.202u {sample_v})
RS src vin 1k
VCLK clk 0 pwl(0 1.2 200n 1.2 202n 0 11.4u 0 11.402u 1.2 11.7u 1.2)
XDUT vin clk vout 0 smp_001_nmos_th
.control
set noaskquit
set numdgt=12
save v(src) v(vin) v(clk) v(vout) i(VCLK) i(VIN)
tran 2n 11.7u 0 2n
meas tran target find v(src) at=180n
meas tran track_v find v(vout) at=180n
meas tran held_v find v(vout) at=220n
meas tran before_input_v find v(vout) at=390n
meas tran after_input_v find v(vout) at=450n
meas tran hold_start_v find v(vout) at=1u
meas tran hold_end_v find v(vout) at=11u
meas tran reacquired_v find v(vout) at=11.65u
let sample_error=abs(v(vout)-target)
meas tran track_error_v max sample_error from=80n to=190n
let pedestal_v=held_v-track_v
let feedthrough_v=after_input_v-before_input_v
let drift_v=hold_end_v-hold_start_v
meas tran reacquire_error_v max sample_error from=11.48u to=11.69u
let hold_drift_v=abs(drift_v)
print hold_drift_v
meas tran hold_error_v max sample_error from=220n to=11u
let input_delta=abs(v(vout)-before_input_v)
meas tran feedthrough_peak_v max input_delta from=400n to=450n
let pedestal_abs_v=abs(pedestal_v)
let supplied=-v(clk)*i(VCLK)
let positive_clock=(supplied+abs(supplied))/2
meas tran clock_energy_j integ positive_clock from=0 to=11.7u
print pedestal_v pedestal_abs_v feedthrough_v drift_v
write transient.raw v(src) v(vin) v(clk) v(vout) i(VCLK) i(VIN)
quit
.endc
.end
