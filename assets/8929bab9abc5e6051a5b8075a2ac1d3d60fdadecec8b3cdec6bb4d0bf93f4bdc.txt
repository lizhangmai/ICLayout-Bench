.include support/models.spice
.include dut.spice
.include parameters.spice
.temp 27
VDD vdd 0 3.3
VP inp 0 PWL(0 1.45 0.1m 1.45 1.1m 1.85 1.2m 1.85 2.2m 1.45 2.3m 1.45)
VN inn 0 1.65
CL out 0 {load_f}
XD vdd out inp inn 0 cmp_001_hyst_diffpair
.control
set numdgt=12
tran 0.2u 2.3m
meas tran rising_input find v(inp) when v(out)=1.65 rise=1
meas tran falling_input find v(inp) when v(out)=1.65 fall=1
let hysteresis_v=rising_input-falling_input
let power=-v(vdd)*i(vdd)
meas tran mean_power_w avg power from=0 to=2.3m
meas tran low_v max v(out) from=0.02m to=0.08m
meas tran high_v min v(out) from=1.12m to=1.18m
print hysteresis_v
write transient.raw time v(inp) v(out) v(vdd) i(vdd)
quit
.endc
.end
