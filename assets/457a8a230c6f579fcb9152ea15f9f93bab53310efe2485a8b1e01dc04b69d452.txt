* GF180MCU nominal characterization: 3.3 V, 27 C, typical models.
.include "support/models.spice"
.include "dut.spice"
.include "parameters.spice"
.temp 27
VDD vdd 0 3.3
IBIAS vdd ib 10u
VP inp 0 DC 1.65 AC .5
VN inn 0 DC 1.65 AC -.5
CLOAD out 0 1p
XDUT vdd out inp inn ib 0 ota_5t
.control
op
let power_w=-v(vdd)*i(vdd)
print power_w
let bias_v=v(out)
print bias_v
ac dec 40 10 1G
let db_gain=vdb(out)
meas ac gain_db find db_gain at=10
meas ac unity_hz when db_gain=0 fall=1
write waveform.raw
quit
.endc
.end
