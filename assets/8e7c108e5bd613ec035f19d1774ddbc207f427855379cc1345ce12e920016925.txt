.include parameters.spice
.options klu reltol=1e-6 abstol=1e-13 vntol=1e-9
.temp 27
VP drivep 0 DC {cm+dm/2} AC 1 PULSE({cm+dm/2} {cm+dm/2+0.1} 1u 10n 10n 9u 20u)
VN driven 0 DC {cm-dm/2} AC {acn} PULSE({cm-dm/2} {cm-dm/2+0.1} 1u 10n 10n 9u 20u)
RP drivep vinp 1000
RN driven vinn 1000
VSS vss 0 0
RL vout 0 {load}
CL vout 0 1p
XDUT vout vinp vinn vss sup_001_vcm_detector
.control
set numdgt=15
set measdgt=15
op
let kcl_residual=abs(i(vp)+i(vn)+i(vss)+v(vout)/@rl[resistance])
if kcl_residual > 1e-10
  echo Error: external DC KCL failed
  quit 1
end
let output_v=v(vout)
let power_w=-v(drivep)*i(vp)-v(driven)*i(vn)
print output_v power_w
write op.raw all
ac dec 40 1 10Meg
let mag=mag(v(vout))
meas ac gain find mag at=1
meas ac gain_10khz find mag at=10k
write ac.raw all
tran 2n 5u 0 2n
linearize
let expected=(op1.v(drivep)+op1.v(driven)+0.2)/2*@rl[resistance]/(@rl[resistance]+500500)
let error=abs(v(vout)-expected)
let first=floor(2u/(time[1]-time[0])+0.5)
let last=floor(4u/(time[1]-time[0])+0.5)
let idx=first
let integral=0
while idx < last
  let integral=integral+(error[idx]+error[idx+1])/2*(time[idx+1]-time[idx])
  let idx=idx+1
end
let step_error_v=integral/2u
print step_error_v
write transient.raw all
quit
.endc
