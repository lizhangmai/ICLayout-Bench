.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.save v(vdd) v(vout) v(vm) v(vp) v(net013) i(VDD) i(VPROBE)
.option klu rshunt=1e12 reltol=1e-5 abstol=1e-12 vntol=1e-8 method=gear maxord=2 itl4=1000
VDD vdd 0 1.2
VIN vp 0 dc {input_v} pulse(0.5 0.7 2u 20n 20n 8u 16u)
IBIAS net013 0 6.86885e-6
VPROBE vm vout dc 0 ac 1
IINJ 0 vm dc 0 ac 0
CLOAD vout 0 {load_f}
XDUT 0 vdd vm vp vout net013 amp_006_leung_dfcfc1
.control
set noaskquit
set numdgt=12
op
let output_v=v(vout)
let output_error_v=abs(v(vout)-v(vp))
let bias_v=v(net013)
let power_w=-v(vdd)*i(VDD)
* Dimensionless units from the paired nominal source operating points.
* Keep the physical observations and independently simulate both anchors.
if v(vp) < 0.6
let power_normalized=power_w/114.7648164198u
let output_error_normalized=output_error_v/2.777053659362m
else
let power_normalized=power_w/114.3276099613u
let output_error_normalized=output_error_v/2.933579773495m
end
print output_v output_error_v bias_v power_w power_normalized output_error_normalized
write op.raw v(vout) v(net013) i(VDD)
ac dec 300 0.01 10g
let bv=-i(VPROBE)
let dv=v(vm)
let oldratio=-v(vout)/v(vm)
write voltage.raw frequency v(vm) v(vout) i(VPROBE)
alter VPROBE acmag=0
alter IINJ acmag=1
ac dec 300 0.01 10g
let aa=-i(VPROBE)
let cc=v(vm)
let bb=ac1.bv
let dd=ac1.dv
let delta=aa*dd-bb*cc
let denominator=1-2*delta+aa-dd
let transfer=(2*delta-aa+dd)/denominator
let forward=(delta-aa)/denominator
let reverse=(delta+dd)/denominator
let gain_db=db(transfer)
let phase_deg=180*cph(transfer)/pi
meas ac dc_gain_db find gain_db at=0.01
meas ac unity_hz when gain_db=0 fall=1
meas ac phase_at_unity find phase_deg when gain_db=0 fall=1
let phase_margin_deg=180+phase_at_unity
print phase_margin_deg
let return_distance=mag(1+transfer)
let minimum_return_distance=vecmin(return_distance)
meas ac hf_gain_db max gain_db from=2g to=10g
let return_phase_excursion_deg=vecmax(abs(180*cph(1+transfer)/pi))
meas ac final_unity_hz when gain_db=0 fall=last
meas ac phase_at_final find phase_deg when gain_db=0 fall=last
let final_phase_margin_deg=180+phase_at_final
print minimum_return_distance return_phase_excursion_deg final_phase_margin_deg
write ac.raw frequency v(vm) v(vout) i(VPROBE) aa bb cc dd transfer forward reverse gain_db phase_deg
reset

tran 2n 18u 0 2n
let error_v=abs(v(vout)-v(vp))
let supply=-v(vdd)*i(VDD)
meas tran recovery_up_v max error_v from=5u to=9.5u
meas tran recovery_down_v max error_v from=13u to=17.5u
meas tran high_v avg v(vout) from=9u to=9.5u
meas tran low_v avg v(vout) from=17u to=17.5u
* Offset-independent finite-window response cost, normalized to the commanded step.
let dynamic_up=abs(v(vout)-high_v)/0.2
let dynamic_down=abs(v(vout)-low_v)/0.2
meas tran response_up_s integ dynamic_up from=2.02u to=9.5u
meas tran response_down_s integ dynamic_down from=10.04u to=17.5u
* Last exit from the 0.1%-of-commanded-step band, interpolated back in.
* A response outside the band at the window end is censored to its duration;
* it remains a valid degraded observation rather than a missing WHEN event.
let sample_index=vector(length(time))
let up_last=vecmax(sample_index*(time ge 2.02u)*(time le 9.5u)*(dynamic_up gt 0.001))
let down_last=vecmax(sample_index*(time ge 10.04u)*(time le 17.5u)*(dynamic_down gt 0.001))
meas tran up_end_error find dynamic_up at=9.5u
meas tran down_end_error find dynamic_down at=17.5u
let settling_up_s=0
if up_end_error > 0.001
let settling_up_s=9.5u-2.02u
else
if up_last > 0
let next_sample=up_last+1
let settling_up_s=time[up_last]+(time[next_sample]-time[up_last])*(dynamic_up[up_last]-0.001)/(dynamic_up[up_last]-dynamic_up[next_sample])-2.02u
end
end
let settling_down_s=0
if down_end_error > 0.001
let settling_down_s=17.5u-10.04u
else
if down_last > 0
let next_sample=down_last+1
let settling_down_s=time[down_last]+(time[next_sample]-time[down_last])*(dynamic_down[down_last]-0.001)/(dynamic_down[down_last]-dynamic_down[next_sample])-10.04u
end
end
print settling_up_s settling_down_s
let step_gain=(high_v-low_v)/0.2
meas tran mean_power_w avg supply from=2u to=18u
meas tran peak_v max v(vout) from=2u to=10u
meas tran trough_v min v(vout) from=10.04u to=18u
print step_gain
write transient.raw v(vout) v(vp) i(VDD)
quit
.endc
.end
