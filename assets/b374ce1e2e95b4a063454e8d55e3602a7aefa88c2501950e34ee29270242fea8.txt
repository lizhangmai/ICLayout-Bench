.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.options reltol=1e-5 abstol=1e-14 vntol=1e-8 method=gear maxord=2
VDD vdd 0 1.2
VREF ref 0 .6
VB1 vb1 0 .5
VB2 vb2 0 .75
VB3 vb3 0 .45
VB4 vb4 0 .55
VFREQ freq 0 {tone_hz}
VAMP amp 0 {signal_v}
VP inp 0 SIN(.6 {signal_v/2} {tone_hz})
VN inn 0 SIN(.6 {-signal_v/2} {tone_hz})
CP outp 0 {load_f}
CN outn 0 {load_f}
VC0 clk_chin 0 pulse(0 1.2 0 50n 50n 99.95u 200u)
VCN0 clk_chin_not 0 pulse(1.2 0 0 50n 50n 99.95u 200u)
VC1 clk_chfb 0 pulse(0 1.2 0 50n 50n 99.95u 200u)
VCN1 clk_chfb_not 0 pulse(1.2 0 0 50n 50n 99.95u 200u)
VC2 clk_chout 0 pulse(0 1.2 0 50n 50n 99.95u 200u)
VCN2 clk_chout_not 0 pulse(1.2 0 0 50n 50n 99.95u 200u)
VC3 clk_chpf 0 pulse(0 1.2 0 50n 50n 99.95u 200u)
VCN3 clk_chpf_not 0 pulse(1.2 0 0 50n 50n 99.95u 200u)

XD inp inn outp outn ref clk_chin clk_chin_not clk_chfb clk_chfb_not clk_chout clk_chout_not clk_chpf clk_chpf_not vdd 0 vb1 vb2 vb3 vb4 inch_p inch_n fbch_p fbch_n g2_p g2_n vsum_p vsum_n pfch_p pfch_n ia_003_fan_chopper_pf
.control
set noaskquit
set num_threads=1
set numdgt=15
save v(inp) v(inn) v(outp) v(outn) v(clk_chin) v(clk_chin_not) i(VP) i(VN) i(VDD) v(inch_p) v(inch_n) v(fbch_p) v(fbch_n) v(g2_p) v(g2_n) v(vsum_p) v(vsum_n) v(pfch_p) v(pfch_n) v(vdd) v(vb1) v(vb2) v(vb3) v(vb4) v(ref) v(freq) v(amp) i(VREF) i(VB1) i(VB2) i(VB3) i(VB4) i(VC0) i(VCN0) v(clk_chfb) v(clk_chfb_not) i(VC1) i(VCN1) v(clk_chout) v(clk_chout_not) i(VC2) i(VCN2) v(clk_chpf) v(clk_chpf_not) i(VC3) i(VCN3)
tran 200n 40m 0 200n
let sample_index=vector(length(time))
meas tran i20 find sample_index at=20m
meas tran i30 find sample_index at=30m
let i40=length(time)-1
let boundary_error_s=abs(time[i20]-20m)+abs(time[i30]-30m)+abs(time[i40]-40m)
if boundary_error_s > 1p
  echo Invalid lock-in integration window
  quit 1
end
let dm=v(outp)-v(outn)
let cm=(v(outp)+v(outn))/2
let ss=sin(2*pi*v(freq)*time)
let cc=cos(2*pi*v(freq)*time)
let ds=dm*ss
let dc=dm*cc
let id=(i(VP)-i(VN))/2
let ids=id*ss
let idc=id*cc
let ds_integral=integ(ds)
let dsmean=(ds_integral[i40]-ds_integral[i20])/(time[i40]-time[i20])
let dc_integral=integ(dc)
let dcmean=(dc_integral[i40]-dc_integral[i20])/(time[i40]-time[i20])
let ids_integral=integ(ids)
let ismean=(ids_integral[i40]-ids_integral[i20])/(time[i40]-time[i20])
let idc_integral=integ(idc)
let icmean=(idc_integral[i40]-idc_integral[i20])/(time[i40]-time[i20])
meas tran dm_mean_v avg dm from=20m to=40m
let amplitude=@VAMP[dc]
let gain_vv=2*sqrt(dsmean^2+dcmean^2)/amplitude
let phase_deg=180*ph(dsmean+j(dcmean))/pi
let zin_ohm=amplitude/(2*sqrt(ismean^2+icmean^2))
let admittance_real_s=-2*ismean/amplitude
let admittance_imag_s=-2*icmean/amplitude
let residual=dm-dm_mean_v-2*dsmean*ss-2*dcmean*cc
meas tran residual_rms_v rms residual from=20m to=40m
meas tran ripple_pp_v pp dm from=20m to=40m
meas tran cm_mean_v avg cm from=20m to=40m
meas tran cm_min_v min cm from=20m to=40m
meas tran cm_max_v max cm from=20m to=40m
let dsfirst=(ds_integral[i30]-ds_integral[i20])/(time[i30]-time[i20])
let dcfirst=(dc_integral[i30]-dc_integral[i20])/(time[i30]-time[i20])
let dslast=(ds_integral[i40]-ds_integral[i30])/(time[i40]-time[i30])
let dclast=(dc_integral[i40]-dc_integral[i30])/(time[i40]-time[i30])
let window_change_v=2*sqrt((dsfirst-dslast)^2+(dcfirst-dclast)^2)
let modulation_error=(v(inch_p)-v(inch_n))*(2*v(clk_chin)/1.2-1)-(v(inp)-v(inn))
let feedback_error=(v(fbch_p)-v(fbch_n))*(2*v(clk_chfb)/1.2-1)-(v(outp)-v(outn))
meas tran modulation_error_rms_v rms modulation_error from=20m to=40m
meas tran feedback_error_rms_v rms feedback_error from=20m to=40m
let pf_error=(v(pfch_p)-v(pfch_n))*(2*v(clk_chpf)/1.2-1)-(v(outp)-v(outn))
meas tran pf_error_rms_v rms pf_error from=20m to=40m
let supplied=-v(vdd)*i(VDD)-v(vb1)*i(VB1)-v(vb2)*i(VB2)-v(vb3)*i(VB3)-v(vb4)*i(VB4)-v(ref)*i(VREF)
let clocks=-1*(v(clk_chin)*i(VC0)+v(clk_chin_not)*i(VCN0)+v(clk_chfb)*i(VC1)+v(clk_chfb_not)*i(VCN1)+v(clk_chout)*i(VC2)+v(clk_chout_not)*i(VCN2)+v(clk_chpf)*i(VC3)+v(clk_chpf_not)*i(VCN3))
meas tran power_w avg supplied from=20m to=40m
meas tran clock_power_w avg clocks from=20m to=40m
print boundary_error_s gain_vv phase_deg zin_ohm admittance_real_s admittance_imag_s window_change_v
write transient.raw v(inp) v(inn) v(outp) v(outn) v(clk_chin) v(clk_chin_not) i(VP) i(VN) i(VDD) v(inch_p) v(inch_n) v(fbch_p) v(fbch_n) v(g2_p) v(g2_n) v(vsum_p) v(vsum_n) v(pfch_p) v(pfch_n)
quit
.endc
.end
