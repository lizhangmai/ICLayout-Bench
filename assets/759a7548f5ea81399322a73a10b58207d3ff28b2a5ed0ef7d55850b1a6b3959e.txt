* Functional pre/post-layout check of the maintained multi-stage TIA core.
* The source title and RF/EM evidence are provenance only.  This deck
* qualifies the redesigned electrical core from 1 MHz through 100 MHz.
.lib /workspace/support/models/cornerHBT.lib hbt_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.lib /workspace/support/models/cornerCAP.lib cap_typ
.include circuit.spice
.param input_current=0
.include parameters.spice
.option rshunt=1e12 reltol=1e-6 vntol=1e-8 abstol=1e-12
.temp 26.85
VSS VSS 0 0
VCC1 VCC1 VSS 1.7
VCC2 VCC2 VSS 2.1
VCC3 VCC3 VSS 2.1
XDUT RFIN RFOUT VCC1 VCC2 VCC3 VSS FMD_QNC_01_LIN_TIA
IIN VSS INJECTED DC {input_current} AC 1
VSENSE INJECTED RFIN 0
CLOAD RFOUT VSS 50f
.control
set numdgt=12
op
let input_bias = v(RFIN)
let output_bias = v(RFOUT)
let supply1_current = -i(VCC1)
let supply2_current = -i(VCC2)
let supply3_current = -i(VCC3)
let supply_power = -v(VCC1)*i(VCC1)-v(VCC2)*i(VCC2)-v(VCC3)*i(VCC3)
print input_bias output_bias supply1_current supply2_current supply3_current supply_power
write operating_point.raw all
ac dec 40 1meg 100meg
let signed_transimpedance = real(v(RFOUT)/i(VSENSE))
let magnitude_transimpedance = mag(v(RFOUT)/i(VSENSE))
meas ac transimpedance_low find signed_transimpedance at=1meg
meas ac transimpedance_high find magnitude_transimpedance at=100meg
write frequency_response.raw all
dc IIN -5u 5u 0.5u
* Measure the dense transfer from the actual VSENSE current.  The scalar
* measurement is part of the public evaluation contract; the acceptance test
* independently recomputes it from dc_transfer.raw.
let dc_i = i(VSENSE)
let dc_o = v(RFOUT)
let dc_i_span = dc_i[length(dc_i)-1]-dc_i[0]
let dc_o_span = dc_o[length(dc_o)-1]-dc_o[0]
let dc_fraction = (dc_i-dc_i[0])/dc_i_span
let dc_line = dc_o[0]+dc_fraction*dc_o_span
let dc_error = abs(dc_o-dc_line)
let dc_error_pct = 100*dc_error/abs(dc_o_span)
meas dc linearity_error_pct MAX dc_error_pct
write dc_transfer.raw all
quit
.endc
.end
