.lib /workspace/support/models/cornerMOSlv.lib mos_tt
.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.save v(vdd) v(vout) v(vp) v(vb1) i(VDD)
.option klu rshunt=1e12 reltol=1e-5 abstol=1e-12 vntol=1e-8 method=gear maxord=2 itl4=1000
VDD vdd 0 1.5
VIN vp 0 dc 0.24 ac 1 pulse(0.24 { 0.24+step_v } 2u 100n 100n 18u 40u)
IBIAS vb1 0 7.11568e-07
XDUT 0 vdd vout vp vout vb1 amp_011_peng_iac
CLOAD vout 0 {load_f}
* Disconnected time marker forces exact integration-window endpoints.
VMARK timing 0 PWL(0 0 15u 0 15.001u 1 20u 1 20.001u 0 35u 0 35.001u 1 40u 1)
.control
set noaskquit
set num_threads=1
set numdgt=15
set measdgt=15
op
let output_v=v(vout)
let bias_v=v(vb1)
let power_w=-v(vdd)*i(VDD)
print output_v bias_v power_w
write op.raw all
ac dec 200 1 1g
let closed_gain_db=db(v(vout))
meas ac closed_gain_10khz_db find closed_gain_db at=10k
write ac.raw all
tran 2n 40u 0 2n
let error_v=abs(v(vout)-v(vp))
let supply=-v(vdd)*i(VDD)
meas tran late_error_v avg error_v from=15u to=20u
meas tran return_error_v avg error_v from=35u to=40u
meas tran high_v avg v(vout) from=15u to=20u
meas tran low_v avg v(vout) from=35u to=40u
meas tran ripple_v pp v(vout) from=15u to=20u
meas tran return_ripple_v pp v(vout) from=35u to=40u
meas tran mean_power_w avg supply from=2u to=40u
meas tran output_min_v min v(vout) from=0 to=40u
meas tran output_max_v max v(vout) from=0 to=40u
write transient.raw v(vout) v(vp) i(VDD) error_v supply
quit
.endc
.end
