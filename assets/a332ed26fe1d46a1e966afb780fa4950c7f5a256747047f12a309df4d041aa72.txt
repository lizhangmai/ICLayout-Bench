.lib /workspace/support/models/cornerMOSlv.lib mos_tt
.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.option rshunt=1e12 reltol=1e-5 abstol=1e-13 vntol=1e-8 method=gear maxord=2
VDD vdd 0 1.2
IBIAS vdd ibias 20u
VCM icm 0 0.5
VSIG sig 0 dc 0 pulse(0 {step_v} 1u 5n 5n 4u 20u)
VREF ref 0 dc 0.75 pulse(0.75 0.8 3u 5n 5n 1u 20u)
EFP fp icm value={(v(sig)-v(outp)+v(outn))/2}
EFN fn icm value={-(v(sig)-v(outp)+v(outn))/2}
VPP inp fp dc 0 ac 0.5
VPN inn fn dc 0 ac -0.5
CLP outp 0 {load_f}
CLN outn 0 {load_f}
IDP 0 outp pulse(0 {kick_a} 6u 5n 5n 50n 20u)
IDN 0 outn pulse(0 {kick_a} 6u 5n 5n 50n 20u)
XDUT inp inn outp outn vdd 0 ibias ref amp_023_fer_fd2s
.control
set noaskquit
set numdgt=12
op
let output_cm_v=(v(outp)+v(outn))/2
let output_dm_v=v(outp)-v(outn)
let bias_v=v(ibias)
let power_w=-v(vdd)*i(VDD)
print output_cm_v output_dm_v bias_v power_w
write op.raw v(outp) v(outn) v(ibias) i(VDD)
ac dec 150 1 1gig
let transfer=(v(outp)-v(outn))/(v(inp)-v(inn))
let gain_db=db(transfer)
let phase_deg=180*cph(transfer)/pi
meas ac dm_gain_db find gain_db at=1
meas ac unity_hz when gain_db=0 fall=1
meas ac phase_at_unity find phase_deg when gain_db=0 fall=1
let phase_margin_deg=180+phase_at_unity
print phase_margin_deg
write dm.raw frequency v(outp) v(outn) v(inp) v(inn) transfer gain_db phase_deg
alter @VPP[acmag]=0
alter @VPN[acmag]=0
alter @VREF[acmag]=1
ac dec 100 1 1gig
let cm_transfer=(v(outp)+v(outn))/2
let cm_gain=mag(cm_transfer)
let cm_db=db(cm_transfer)
meas ac cm_gain_vv find cm_gain at=1
meas ac cm_peak_vv max cm_gain
write cm.raw frequency v(outp) v(outn) cm_transfer
tran 1n 8u 0 1n
let dm=v(outp)-v(outn)
let cm=(v(outp)+v(outn))/2
let dm_err=abs(dm-v(sig))
let cm_err=abs(cm-v(ref))
let supplied=-v(vdd)*i(VDD)
meas tran dm_high_v avg dm from=1.8u to=1.95u
meas tran dm_up_error_v max dm_err from=1.5u to=1.95u
meas tran dm_down_error_v max dm_err from=5.5u to=5.95u
meas tran cm_high_v avg cm from=3.8u to=3.95u
meas tran cm_up_error_v max cm_err from=3.5u to=3.95u
meas tran cm_down_error_v max cm_err from=4.5u to=5.9u
meas tran cm_kick_error_v max cm_err from=6.5u to=7.9u
meas tran cm_kick_peak_v max cm_err from=6u to=6.5u
let dm_abs=abs(dm)
meas tran dm_peak_v max dm_abs from=1u to=5u
meas tran dm_cm_error_v max dm_err from=3.5u to=3.95u
meas tran cm_dm_error_v max cm_err from=1.5u to=1.95u
meas tran mean_power_w avg supplied from=0 to=8u
write transient.raw v(outp) v(outn) v(sig) v(ref) i(VDD)
quit
.endc
.end
