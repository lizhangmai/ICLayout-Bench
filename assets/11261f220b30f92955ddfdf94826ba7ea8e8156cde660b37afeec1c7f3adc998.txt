* Regulator nominal and finite-window perturbation observations.
.include support/models.spice
.include dut.spice
.include parameters.spice
.temp 27
.save v(vdd) v(vss) v(vout) v(cout) v(vref) i(vdd) i(vss) i(vref) v(vb1) v(vb2) i(vb1) i(vb2) @iload[current]
VSS vss 0 0
.options reltol=1e-6 abstol=1e-13 vntol=1e-9
VDD vdd 0 dc {supply_v} ac 1 pulse({supply_v} {supply_v+line_step_v} 100u 1u 1u 100u 500u)
VB1 vb1 0 1
VB2 vb2 0 .025
VREF vref 0 1.8
ILOAD vout 0 dc {load_a} pulse({load_a} {load_a+load_step_a} 100u 1u 1u 100u 500u)
RESR vout cout .1
COUT cout 0 1n
XDUT vdd vout vss vb1 vb2 vref ldo_002_analoggym_folded_cascode
.csparam do_static={do_static}
.control
set numdgt=15
set measdgt=15
if $&do_static
op
let kcl_a=abs(i(vdd)+i(vss)+i(vref)+@iload[current]+i(vb1)+i(vb2))
print kcl_a
let output_v=v(vout)
let regulation_error_v=abs(v(vout)-1.8)
let power_w=-v(vdd)*i(vdd)
print output_v regulation_error_v power_w
write op.raw all
ac dec 100 1 100Meg
let rejection=-db(v(vout))
meas ac psrr_db find rejection at=1k
write ac.raw all
end
tran 10n 400u 0 10n
let err=abs(v(vout)-1.8)
let supply_power=-v(vdd)*i(vdd)
meas tran quiet_v avg v(vout) from=80u to=100u
meas tran quiet_ripple_v pp v(vout) from=80u to=100u
meas tran high_v avg v(vout) from=180u to=200u
meas tran recovery_v avg v(vout) from=380u to=400u
meas tran recovery_ripple_v pp v(vout) from=380u to=400u
meas tran excursion_v pp v(vout) from=100u to=400u
meas tran late_error_v avg err from=380u to=400u
meas tran mean_power_w avg supply_power from=0 to=400u
write transient.raw all
quit
.endc
.end
