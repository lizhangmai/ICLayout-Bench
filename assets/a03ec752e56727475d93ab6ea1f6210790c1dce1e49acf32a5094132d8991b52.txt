.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.save v(vdd) v(vb1) v(vb2) v(vb3) v(ref) v(icm) v(sig) v(inp) v(inn) v(fp) v(fn) v(outp) v(outn) v(s1p) v(s1n) v(ctl) i(VDD) i(VB1) i(VB2) i(VB3) i(VREF) v(vb4) i(VB4)
.options rshunt=1e12 reltol=1e-6 abstol=1e-14 vntol=1e-9 method=gear maxord=2
VDD vdd 0 1.2
VB1 vb1 0 .3983759766
VB2 vb2 0 .65
VB3 vb3 0 .45
VB4 vb4 0 .55
VREF ref 0 pulse(.5 {.5+ref_step} 50u 20n 20n 15u 200u)
VSIG sig 0 dc 0 ac 0 pulse(0 {step_v} 10u 20n 20n 10u 200u)
VCM icm 0 pulse(.6 {.6+icm_step} 75u 20n 20n 15u 200u)
EFP fp icm value={(v(sig)-v(outp)+v(outn))/2}
EFN fn icm value={-(v(sig)-v(outp)+v(outn))/2}
LP fp bp 1T
CPB bp 0 1
VP inp bp dc 0 ac .5
LN fn bn 1T
CNB bn 0 1
VN inn bn dc 0 ac .5 180
CP outp 0 {load_f}
CN outn 0 {load_f}
IP 0 outp pulse(0 {kick_a} 30u 20n 20n 2u 200u)
IN 0 outn pulse(0 {kick_a} 30u 20n 20n 2u 200u)
XDUT inp inn outp outn vdd 0 vb1 vb2 vb3 vb4 ref s1p s1n ctl amp_034_fan_chopper_cmfb
.control
set noaskquit
set num_threads=1
set numdgt=15
set measdgt=15
op
let cm_bias_v=(v(outp)+v(outn))/2
let s1_bias_v=(v(s1p)+v(s1n))/2
let dm_bias_v=v(outp)-v(outn)
let power_w=-v(vdd)*i(VDD)-v(vb1)*i(VB1)-v(vb2)*i(VB2)-v(vb3)*i(VB3)-v(ref)*i(VREF)-v(vb4)*i(VB4)
print cm_bias_v s1_bias_v dm_bias_v power_w
write op.raw all
let dc_feedback_error_v=max(abs(v(inp)-v(fp)),abs(v(inn)-v(fn)))
print dc_feedback_error_v
if dc_feedback_error_v > 1e-6
 echo Error: invalid DC feedback
 quit 1
end
ac dec 200 1 1g
let transfer=(v(outp)-v(outn))/(v(inp)-v(inn))
let gain=db(transfer)
let fixture_cm_max=vecmax(mag((v(inp)+v(inn))/2))
meas ac gain_db find gain at=1
meas ac gain_10khz_db find gain at=10k
print fixture_cm_max
if fixture_cm_max > 1e-6
 echo Error: invalid balanced AC excitation
 quit 1
end
write ac.raw all
quit
.endc
.end
