.include support/models.spice
.include dut.spice
.include parameters.spice
.temp 27
VDD vdd 0 DC {supply_v}
IB vdd ibias 20u
VCM vcm 0 {common_v}
VD diff 0 DC 0 AC 1
EP vinp vcm diff 0 0.5
EN vinn vcm diff 0 -0.5
CLP voutp 0 1p
CLN voutn 0 1p
XDUT vdd voutp voutn vinp vinn ibias 0 dp_001_resistive_load
.control
set noaskquit
set numdgt=12
op
let output_v=(v(voutp)+v(voutn))/2
let imbalance_v=abs(v(voutp)-v(voutn))
let bias_v=v(ibias)
let power_w=-v(vdd)*i(vdd)
print output_v imbalance_v bias_v power_w
write op.raw v(vdd) v(voutp) v(voutn) v(ibias) i(vdd)
ac dec 100 1 1G
let gain_db=db(v(voutn)-v(voutp))
meas ac gain_db find gain_db at=10
let drop_db=db(v(voutn)-v(voutp))-gain_db
meas ac bandwidth_hz when drop_db=-3 fall=1
write ac.raw frequency v(voutn) v(voutp) v(diff)
dc VD -0.0101 0.0101 0.0001
let vdout=v(voutn)-v(voutp)
meas dc low_v find vdout at=-0.01
meas dc high_v find vdout at=0.01
let slope=(high_v-low_v)/0.02
let error_v=abs(vdout-(low_v+slope*(v(diff)+0.01)))
meas dc linearity_v max error_v from=-0.01 to=0.01
print slope
write transfer.raw v(diff) v(voutn) v(voutp)
quit
.endc
.end
