.include support/models.spice
.include dut.spice
.include parameters.spice
.temp {temperature_c}
.option method=gear maxord=2
VDD vdd 0 DC {supply_v} PWL(0 0 {rise_s} {supply_v})
CL vout 0 100f
XDUT vdd vout 0 tsn_002_ptat_classic
.save v(vdd) v(vout) i(vdd)
.control
set noaskquit
set numdgt=12
op
let output_v=v(vout)
let power_w=-v(vdd)*i(vdd)
print output_v power_w
write op.raw all
dc temp -20 100 1
let vv=v(vout)
let pp=-v(vdd)*i(vdd)
meas dc cold_v find vv at=-20
meas dc hot_v find vv at=100
let slope_v_per_c=(hot_v-cold_v)/120
let error_v=abs(vv-(cold_v+slope_v_per_c*vector(length(vv))))
meas dc curvature_v max error_v
let local_slope=deriv(vv)
meas dc minimum_slope min local_slope
meas dc maximum_slope max local_slope
meas dc peak_power_w max pp
print slope_v_per_c
write temperature.raw all
reset
tran 1n 100u 0 1n uic
let vv=v(vout)
let error=abs(vv-op1.output_v)
meas tran startup_error_v max error from=90u to=100u
meas tran late_min min vv from=90u to=100u
meas tran late_max max vv from=90u to=100u
let ripple_v=late_max-late_min
let pp=-v(vdd)*i(vdd)
meas tran final_power_w avg pp from=90u to=100u
print ripple_v
write startup.raw all
quit
.endc
.end
