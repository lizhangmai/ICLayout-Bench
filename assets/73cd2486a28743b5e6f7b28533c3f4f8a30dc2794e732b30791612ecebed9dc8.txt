* Source and extracted-netlist calibration deck.  The measured RF scope is
* 100 MHz small-signal voltage gain; the historical 160 GHz claims are not
* inherited by this maintained functional case.
.lib /workspace/support/models/cornerHBT.lib hbt_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include circuit.spice
.option rshunt=1e12 reltol=1e-6 vntol=1e-8 abstol=1e-12
.temp 27
VDD VDD 0 DC 1.2 AC 0
VSS VSS 0 DC 0 AC 0
VBIAS VBIAS 0 DC {vbias_dc} AC 0
VIN IN 0 DC {vin_dc} AC 1
CLOAD OUT 0 50f
XDUT IN OUT VDD VSS VBIAS LNA160_FOUR_STAGE
.control
set noaskquit
set numdgt=12
op
let vin_bias=v(in)
let out_bias=v(out)
let i_vdd=-i(vdd)
let i_vbias=-i(vbias)
echo vin_bias = $&vin_bias
echo out_bias = $&out_bias
echo i_vdd = $&i_vdd
echo i_vbias = $&i_vbias
write operating_point.raw all
ac dec 40 1meg 1gig
let gain_db=db(v(out)/v(in))
meas ac gain_db find gain_db at=100meg
write frequency_response.raw all
quit
.endc
.end
