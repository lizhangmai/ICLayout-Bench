* OTA open-loop AC with DC unity feedback; nominal TT, 27 C.
.lib /workspace/support/models/cornerMOSlv.lib mos_tt
.lib /workspace/support/models/cornerRES.lib res_typ
.include dut.spice
.temp 27
.option rshunt=1e12
VDD vdd 0 1.5
VIN vinp 0 DC 0.8 AC 1
IBIAS vdd ibias DC 20u
LFB vout vinn 4G
CFB vinn 0 4G
CL vout 0 10p
XDUT vdd vout vinp vinn ibias 0 amp_001_5t
.control
set noaskquit
set numdgt=12
op
let output_bias = v(vout)
let bias_voltage = v(ibias)
let supply_power = -v(vdd)*i(VDD)
print output_bias bias_voltage supply_power
write ota-op.raw all
ac dec 100 1 100meg
let transfer = v(vout)/(v(vinp)-v(vinn))
let gain_db = db(transfer)
let phase_deg = 180*cph(transfer)/pi
meas ac low_frequency_gain find gain_db at=1
meas ac unity_gain_bandwidth when gain_db=0 fall=1
meas ac phase_at_unity find phase_deg when gain_db=0 fall=1
let phase_margin = 180+phase_at_unity
print phase_margin
write ota-ac.raw all
quit
.endc
.end
