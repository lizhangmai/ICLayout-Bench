VA port_a 0 {common_v+.005*direction}
VB port_b 0 {common_v-.005*direction}
VCTL vctl 0 1.5
VNOT vctl_not 0 0
XDUT port_a port_b vctl vctl_not vdd vss sw_001_transmission_gate_pair
.control

let on_current_a=abs(i(va))
let ron_ohm=abs((v(port_a)-v(port_b))/i(va))
let kcl_a=abs(i(va)+i(vb)+i(vdd)+i(vss)+i(vctl)+i(vnot))

if kcl_a > 1e-09
  echo Invalid DC current conservation: kcl_a exceeds 1e-09 A
  quit 1
end
print on_current_a ron_ohm kcl_a
write op.raw all
alter VCTL 0
alter VNOT 1.5

let leakage_a=abs(i(va))
print leakage_a
write off.raw all
