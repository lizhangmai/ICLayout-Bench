.lib /workspace/support/models/cornerMOSlv.lib mos_tt
.lib /workspace/support/models/cornerCAP.lib cap_typ
.lib /workspace/support/models/cornerRES.lib res_typ
.include parameters.spice
.include dut.spice
.temp 27
.save v(vdd) v(vout) v(vm) v(vp) v(vb1) i(VDD) i(VPROBE)
.option klu rshunt=1e12 reltol=1e-5 abstol=1e-12 vntol=1e-8 method=gear maxord=2 itl4=1000
VDD vdd 0 1.2
VIN vp 0 dc {input_v} pulse(0.5 0.7 2u 20n 20n 18u 100u)
IBIAS vb1 0 1.6231125e-07
VPROBE vm vout dc 0 ac 1
IINJ 0 vm dc 0 ac 0
CLOAD vout 0 {load_f}
XDUT 0 vdd vm vp vout vb1 amp_012_peng_tcfc
.control
set noaskquit
set numdgt=12
op
let output_v=v(vout)
let output_error_v=abs(v(vout)-v(vp))
let bias_v=v(vb1)
let power_w=-v(vdd)*i(VDD)
print output_v output_error_v bias_v power_w
write op.raw v(vout) v(vb1) i(VDD)
ac dec 300 0.01 10g
let bv=-i(VPROBE)
let dv=v(vm)
let oldratio=-v(vout)/v(vm)
write voltage.raw frequency v(vm) v(vout) i(VPROBE)
alter VPROBE acmag=0
alter IINJ acmag=1
ac dec 300 0.01 10g
let aa=-i(VPROBE)
let cc=v(vm)
let bb=ac1.bv
let dd=ac1.dv
let delta=aa*dd-bb*cc
let denominator=1-2*delta+aa-dd
let transfer=(2*delta-aa+dd)/denominator
let forward=(delta-aa)/denominator
let reverse=(delta+dd)/denominator
let gain_db=db(transfer)
let phase_deg=180*cph(transfer)/pi
meas ac dc_gain_db find gain_db at=0.01
meas ac unity_hz when gain_db=0 fall=1
meas ac phase_at_unity find phase_deg when gain_db=0 fall=1
let phase_margin_deg=180+phase_at_unity
print phase_margin_deg
let return_distance=mag(1+transfer)
let minimum_return_distance=vecmin(return_distance)
meas ac hf_gain_db max gain_db from=2g to=10g
let return_phase_excursion_deg=vecmax(abs(180*cph(1+transfer)/pi))
meas ac final_unity_hz when gain_db=0 fall=last
meas ac phase_at_final find phase_deg when gain_db=0 fall=last
let final_phase_margin_deg=180+phase_at_final
print minimum_return_distance return_phase_excursion_deg final_phase_margin_deg
write ac.raw frequency v(vm) v(vout) i(VPROBE) aa bb cc dd transfer forward reverse gain_db phase_deg
reset

tran 2n 40u 0 2n
let error_v=abs(v(vout)-v(vp))
let supply=-v(vdd)*i(VDD)
meas tran recovery_up_v max error_v from=15u to=19u
meas tran recovery_down_v max error_v from=30u to=39u
meas tran high_v avg v(vout) from=18u to=19u
meas tran low_v avg v(vout) from=38u to=39u
let step_gain=(high_v-low_v)/0.2
meas tran mean_power_w avg supply from=2u to=40u
meas tran peak_v max v(vout) from=2u to=20u
meas tran trough_v min v(vout) from=20.04u to=40u
print step_gain
write transient.raw v(vout) v(vp) i(VDD)
quit
.endc
.end
