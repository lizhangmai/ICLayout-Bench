.include dut.spice
.include parameters.spice
.temp 27
.options reltol=1e-6 abstol=1e-13 vntol=1e-9
VDD vdd 0 2
VSS vss 0 0
VB vb 0 1.2
VREF vref 0 1.8
ILOAD vout 0 1m
RESR vout cout .1
COUT cout 0 50p
XDUT vdd vout vss vb vref ldo_003_analoggym_simple
.save v(vdd) v(vss) v(vout) v(cout) v(vb) v(vref) i(vdd) i(vss) i(vb) i(vref)
.control
set numdgt=15
set measdgt=15
dc VDD 1.9 2.2 .005
meas dc low_v find v(vout) at=2
meas dc high_v find v(vout) at=2.1
let line_reg_v_per_v=abs((v(vout)[40]-v(vout)[20])/(v(vdd)[40]-v(vdd)[20]))
print line_reg_v_per_v
write line.raw all
dc ILOAD .001 .005 .0001
meas dc light_v find v(vout) at=.001
meas dc heavy_v find v(vout) at=.005
let load_reg_ohm=abs((v(vout)[length(v(vout))-1]-v(vout)[0])/.004)
print load_reg_ohm
write load.raw all
quit
.endc
